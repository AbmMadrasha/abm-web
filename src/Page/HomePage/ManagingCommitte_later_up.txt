import React, { useRef, useState, useEffect } from 'react'

const ManagingCommitte = () => {

  const [managingMember, setManagingMember] = useState([])

  // Fetch API
  const fetchApi = async () => {
    try {
      const response = await fetch(
        //'https://oracleapex.com/ords/abm_ws/Module/TEACHER/'
      )

      const data = await response.json()

      console.log(data)

      setManagingMember(data.items)

    } catch (error) {
      console.log('Fetch Error:', error)
    }
  }

  // Load data
  useEffect(() => {
    fetchApi()
  }, [])

  // Duplicate for infinite slider
  const loopData = [...managingMember, ...managingMember]

  const sliderRef = useRef(null)

  const [isDown, setIsDown] = useState(false)
  const [startX, setStartX] = useState(0)
  const [scrollLeft, setScrollLeft] = useState(0)
  const [isHovered, setIsHovered] = useState(false)

  // Auto Scroll
  useEffect(() => {

    const slider = sliderRef.current

    const interval = setInterval(() => {

      if (!slider || isHovered || isDown) return

      slider.scrollLeft += 1

      if (slider.scrollLeft >= slider.scrollWidth / 2) {
        slider.scrollLeft -= slider.scrollWidth / 2
      }

    }, 20)

    return () => clearInterval(interval)

  }, [isHovered, isDown])

  // Drag Scroll
  const handleMouseDown = (e) => {
    setIsDown(true)
    setStartX(e.pageX - sliderRef.current.offsetLeft)
    setScrollLeft(sliderRef.current.scrollLeft)
  }

  const handleMouseLeave = () => {
    setIsDown(false)
    setIsHovered(false)
  }

  const handleMouseUp = () => setIsDown(false)

  const handleMouseMove = (e) => {
    if (!isDown) return

    e.preventDefault()

    const x = e.pageX - sliderRef.current.offsetLeft

    const walk = (x - startX) * 2

    sliderRef.current.scrollLeft = scrollLeft - walk
  }

  return (
    <div className='px-6 md:px-10 py-10'>

      <div className='text-center text-black text-2xl font-bold py-5'>
        Teacher List
      </div>

      <div
        ref={sliderRef}
        className='
          flex gap-4 overflow-x-auto cursor-grab active:cursor-grabbing
          [scrollbar-width:none] [-ms-overflow-style:none]
          [&::-webkit-scrollbar]:hidden
        '
        onMouseDown={handleMouseDown}
        onMouseLeave={handleMouseLeave}
        onMouseUp={handleMouseUp}
        onMouseMove={handleMouseMove}
        onMouseEnter={() => setIsHovered(true)}
      >

        {loopData.map((item, index) => (

          <div key={index} className='shadow-md min-w-62.5'>

            <div className='flex flex-col p-6 justify-center items-center border-2 border-gray-100'>

              <div className='h-24 w-24 rounded-full bg-[#1B4962] text-white border-4 border-amber-600 flex items-center justify-center'>
                {item.status}
              </div>

              <div className='text-[12px] text-black mt-4'>
                {item.teacher_name}
              </div>

              <div>
                 {item.teacher_designation}
              </div>

            </div>

          </div>

        ))}

      </div>

    </div>
  )
}

export default ManagingCommitte
